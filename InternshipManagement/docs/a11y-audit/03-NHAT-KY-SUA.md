Kiểm tra computed style thủ công trong DevTools trên `/companies/new`:

```js
const button = [...document.querySelectorAll('button')].find((item) => item.innerText.includes('Quay lại'))
const color = getComputedStyle(button.querySelector('.p-button-label')).color
let surface = button
let background = 'transparent'
while (surface && background === 'transparent') {
  background = getComputedStyle(surface).backgroundColor
  if (background === 'rgba(0, 0, 0, 0)') background = 'transparent'
  if (background === 'transparent') surface = surface.parentElement
}
const rgb = (value) => value.match(/[\d.]+/g).slice(0, 3).map(Number)
const luminance = (value) => rgb(value).map((channel) => channel / 255).map((channel) => channel <= 0.04045 ? channel / 12.92 : ((channel + 0.055) / 1.055) ** 2.4).reduce((sum, channel, index) => sum + channel * [0.2126, 0.7152, 0.0722][index], 0)
const [lighter, darker] = [luminance(color), luminance(background)].sort((a, b) => b - a)
console.log({ color, background, contrast: (lighter + 0.05) / (darker + 0.05) })
```

Output sau sửa: `{ color: "rgb(95, 101, 95)", background: "rgb(246, 244, 238)", contrast: 5.43 }`.
